# board-discovery probes: one per line, "name|timeout s|diff mode|command".
# Every command is read-only, POSIX sh / BusyBox safe, one line (serial consoles cut long lines at ~4 KB).
# diff modes (for `discover --diff`): full = line diff, sort = sorted unique lines, skip = volatile,
#   cmdonly = drop the first two fields (pid, rss) then sort, lastcol = compare only the last field (sizes and dates change).
identity|15|full|hostname; uname -srm; cat /etc/os-release 2>/dev/null; for f in /proc/device-tree/model /proc/device-tree/compatible /proc/device-tree/serial-number /etc/machine-id /etc/issue /etc/version /etc/build /etc/timestamp; do [ -r $f ] && printf '%s: %s\n' "$f" "$(tr '\0' ' ' <$f | head -n 3)"; done
clock|10|skip|date; uptime; cat /etc/timezone 2>/dev/null; readlink /etc/localtime; timedatectl 2>/dev/null | head -n 8
boot|10|full|cat /proc/cmdline; ls /boot 2>/dev/null | head -n 20; command -v fw_printenv >/dev/null && fw_printenv 2>/dev/null | grep -iE '^(boot_?part|bootslot|slot|bootcount|upgrade|rauc|mender|boot_order)' | head -n 10
cpu-mem|10|full|grep -m6 -E '^(model name|Hardware|Processor|CPU part|CPU implementer|Revision|Serial)' /proc/cpuinfo; grep -c ^processor /proc/cpuinfo; grep -E 'MemTotal|SwapTotal' /proc/meminfo
storage|15|sort|cat /proc/partitions; ls -l /dev/disk/by-label /dev/disk/by-partlabel 2>/dev/null | grep -- '->'

timers-cron|15|sort|systemctl list-timers --all --no-legend --no-pager 2>/dev/null | awk '{for(i=1;i<=NF;i++) if ($i ~ /\.timer$/) print $i, $(i+1)}'; ls /etc/cron* /var/spool/cron/crontabs 2>/dev/null
processes|20|cmdonly|me=$(cut -d' ' -f6 /proc/self/stat); for d in /proc/[0-9]*; do [ "$(cut -d' ' -f6 $d/stat 2>/dev/null)" = "$me" ] && continue; c=$(tr '\0' ' ' 2>/dev/null <$d/cmdline); [ -z "$c" ] && continue; echo "${d#/proc/} $(awk '/^VmRSS/{print $2}' $d/status 2>/dev/null) $c"; done | sort -k2 -nr | head -n 70 | cut -c1-300
nonstd-exes|20|sort|for d in /proc/[0-9]*; do readlink $d/exe 2>/dev/null; done | sort -u | grep -vE '^/(usr/)?(s?bin|lib(exec|32|64)?)/'
listening|30|sort|S=$(for d in /proc/[0-9]*; do ls -l $d/fd 2>/dev/null | sed -n "s/.*socket:\[\([0-9]*\)\]/\1 ${d#/proc/}/p"; done); for f in /proc/net/tcp /proc/net/tcp6 /proc/net/udp /proc/net/udp6; do awk -v f=${f##*/} 'NR>1 && (($4=="0A" && f ~ /tcp/) || ($4=="07" && f ~ /udp/)) {print f, $2, $10}' $f 2>/dev/null; done | while read f a i; do p=$(echo "$S" | awk -v i=$i '$1==i {print $2; exit}'); h=${a%:*}; case $h in 00000000|00000000000000000000000000000000) h='*';; 0100007F|00000000000000000000000001000000) h=lo;; esac; echo "$f $h:$(printf %d 0x${a##*:}) ${p:+$(cat /proc/$p/comm 2>/dev/null)}"; done | sort -u
